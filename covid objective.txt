# Analysis  the Birth rate, crude (per 1,000 people) vs Death rate, crude (per 1,000 people) using Hist Plot

sns.set(style="whitegrid")

# Drop rows with missing values in either Birth or Death rate columns
filtered_df = df.dropna(subset=["Birth rate, crude (per 1,000 people)","Death rate, crude (per 1,000 people)"])

# Create histograms for birth and death rates
plt.figure(figsize=(12, 6))

# Histogram for Birth Rate
sns.histplot(filtered_df["Birth rate, crude (per 1,000 people)"], color="skyblue", label="Birth Rate", kde=True)

# Histogram for Death Rate
sns.histplot(filtered_df["Death rate, crude (per 1,000 people)"], color="salmon", label="Death Rate", kde=True)

plt.title("Distribution of Crude Birth and Death Rates (per 1,000 people)")
plt.xlabel("Rate per 1,000 People")
plt.ylabel("Frequency")
plt.legend()
plt.tight_layout()
plt.show()
